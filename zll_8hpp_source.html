413template < typename T, typename Acc, typename Compare = std::less<> >
414requires( _provides_ll_header< T, Acc > )
415std::pair< T*, T* >
416merge_ranges( T& lhf, T& lhl, T& rhf, T& rhl, Compare&& comp = std::less<>{} ) noexcept(
417 _nothrow_access< Acc, T > && noexcept( comp( lhf, rhf ) ) )
418{
419 _ll_hdr& ll = Acc::get( lhl );
420 _ll_hdr& rf = Acc::get( rhf );
421 _ll_hdr& rl = Acc::get( rhl );
422 _unlink( rf, rl );
423 _ll_hdr* lh = &Acc::get( lhf );
424 _ll_hdr* rh = &rf;
425 _ll_hdr* first = nullptr;
426 _ll_hdr* last = nullptr;
427 _ll_word pred = lh->_prev;
428 _ll_word succ = ll._next;
429 while ( lh && rh ) {
430 _ll_hdr* tmp = nullptr;
431 if ( comp( _node< T, Acc >( *rh ), _node< T, Acc >( *lh ) ) ) {
432 tmp = rh;
433 rh = rh->_next.a();
434 } else {
435 tmp = lh;
436 lh = lh == &ll ? nullptr : lh->_next.a();
437 }
438 _unlink( *tmp, *tmp );
439 if ( last )
440 _link( *tmp, *tmp, *last, last->_next );
441 else
442 first = tmp;
443 last = tmp;
444 }
445 ZLL_ASSERT( first );
446 ZLL_ASSERT( last );
447 if ( lh ) {
448 last->_next = *lh;
449 lh->_prev = *last;
450 last = &ll;
451 } else if ( rh ) {
452 last->_next = *rh;
453 rh->_prev = *last;
454 last = &rl;
455 }
456 _link( *first, *last, pred, succ );
457
458 return { &_node< T, Acc >( *first ), &_node< T, Acc >( *last ) };

532 _nothrow_access< Acc, T > && noexcept( cmp( first, last ) ) )
533{

544 _unlink( *n, *n );
